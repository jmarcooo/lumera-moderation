        // --- WARNING MODAL LOGIC FOR GOING BACK ---
        function attemptGoBack() {
            if (currentView === 'detail') {
                document.getElementById('warning-modal').style.display = 'flex';
            } else {
                showList();
            }
        }

            document.getElementById('warning-modal').style.display = 'none';
            showList();

            document.getElementById('warning-modal').style.display = 'none';

        // --- IMAGE VIEWER MODAL LOGIC ---
        let currentScale = 1;
        let currentRotate = 0;
        let currentFlipX = 1;
        let currentFlipY = 1;
        const modalEl = document.getElementById('image-modal');
        const modalImgTarget = document.getElementById('modal-img-target');

        function openImageModal(src) {
            currentScale = 1;
            currentRotate = 0;
            currentFlipX = 1;
            currentFlipY = 1;
            applyTransform();

            modalImgTarget.src = src;
            modalEl.style.display = 'flex';
        }

        function closeImageModal() {
            modalEl.style.display = 'none';
            modalImgTarget.src = '';
        }

        function applyTransform() {
            modalImgTarget.style.transform = `scale(${currentScale}) rotate(${currentRotate}deg) scaleX(${currentFlipX}) scaleY(${currentFlipY})`;
        }

        function modifyImage(action) {
            switch(action) {
                case 'zoom-in': currentScale += 0.2; break;
                case 'zoom-out': currentScale = Math.max(0.2, currentScale - 0.2); break;
                case 'rotate-left': currentRotate -= 90; break;
                case 'rotate-right': currentRotate += 90; break;
                case 'flip-h': currentFlipX *= -1; break;
                case 'flip-v': currentFlipY *= -1; break;
            }
            applyTransform();
        }

        // Mouse Wheel Zoom
        modalEl.addEventListener('wheel', function(e) {
            if (modalEl.style.display === 'flex') {
                e.preventDefault();
                if (e.deltaY < 0) {
                    modifyImage('zoom-in');
                } else {
                    modifyImage('zoom-out');
                }
            }
        }, { passive: false });

        modalEl.addEventListener('click', function(e) {
            if (e.target === modalEl) closeImageModal();
        });

        document.addEventListener('keydown', function(e) {
            if (modalEl.style.display === 'flex') {
                if (e.key === 'Escape' || e.key === 'Esc') {
                    closeImageModal();
                }
            }
        });
    </script>
